broker/src/check.ts
// Runs the checks in order. The first failed check stops everything (early return).
function runChecks(token, action) {
  const mandate = mandates.get(token);
  if (mandate === undefined) return deny("unknown_token", "unknown token");
  if (mandate.status !== "approved") return deny("not_approved", …);
  if (Date.now() > mandate.expiresAt) return deny("expired", "mandate expired");
  if (mandate.revoked) return deny("revoked", "mandate revoked");
  if (!mandate.allowedActions.includes(action)) return deny("action_not_allowed", …);
  if (mandate.used >= mandate.limit) return deny("limit_reached", …);

  mandate.used += 1;        // only allowed actions use up the limit
  return { allowed: true, code: "ok", reason: "ok" };   // "yes" only at the very end
}
